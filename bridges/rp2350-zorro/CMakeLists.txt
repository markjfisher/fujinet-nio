cmake_minimum_required(VERSION 3.21)
set(BRIDGE_MODE "host" CACHE STRING "host or firmware or stimulus")
option(DUT_CAPTURE_W1 "Build the feasibility DUT with W1 16-bit selected-write capture" OFF)
if(NOT BRIDGE_MODE MATCHES "^(host|firmware|stimulus)$")
    message(FATAL_ERROR "BRIDGE_MODE must be host or firmware or stimulus")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_LIST_DIR}/dependencies.json")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/.deps")
if(BRIDGE_MODE STREQUAL "firmware")
    if(NOT PICO_BOARD STREQUAL "waveshare_core2350b" OR NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "Firmware requires waveshare_core2350b / rp2350-arm-s (RP2350B)")
    endif()
endif()
if(BRIDGE_MODE STREQUAL "stimulus")
    if(NOT PICO_BOARD STREQUAL "pico" OR NOT PICO_PLATFORM STREQUAL "rp2040")
        message(FATAL_ERROR "Stimulus requires pico / rp2040 with RAM-only binary")
    endif()
endif()
if(NOT BRIDGE_MODE STREQUAL "host")
    if(NOT DEFINED PICO_SDK_PATH)
        if(DEFINED ENV{PICO_SDK_PATH})
            set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
        else()
            set(PICO_SDK_PATH "${DEPS}/pico-sdk")
        endif()
    endif()
    set(SDK_ARGS --sdk-path "${PICO_SDK_PATH}")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/scripts/check_pio_policy.py"
    RESULT_VARIABLE POLICY_RESULT)
if(NOT POLICY_RESULT EQUAL 0)
    message(FATAL_ERROR "First-party PIO policy failed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/scripts/bootstrap.py"
    --mode "${BRIDGE_MODE}" --check ${SDK_ARGS} RESULT_VARIABLE DEPENDENCY_RESULT)
if(NOT DEPENDENCY_RESULT EQUAL 0)
    message(FATAL_ERROR "Dependency validation failed; see bootstrap repair guidance above")
endif()
if(BRIDGE_MODE STREQUAL "host")
    project(rp2350_zorro_host C)
    include(cmake/validation.cmake)
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Host tests require a native compiler")
    endif()
    enable_testing()
    include(cmake/host.cmake)
else()
    include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    project(rp2350_zorro_firmware C CXX ASM)
    include(cmake/validation.cmake)
    include(cmake/picotool.cmake)
    add_dependencies(bridge_picotool_build bridge_validate)
    pico_sdk_init()
    if(BRIDGE_MODE STREQUAL "stimulus")
        include(cmake/stimulus.cmake)
    else()
        if(NOT PICO_RP2350 OR PICO_RP2350A)
            message(FATAL_ERROR "RP2350B package required")
        endif()
        add_executable(bridge_capture src/main.c src/capture_program.c)
        add_dependencies(bridge_capture bridge_validate)
        target_include_directories(bridge_capture PRIVATE src "${DEPS}/apio/include")
        target_link_libraries(bridge_capture PRIVATE pico_stdlib hardware_pio)
        pico_enable_stdio_uart(bridge_capture 0)
        pico_enable_stdio_usb(bridge_capture 0)
        pico_add_extra_outputs(bridge_capture)
        add_executable(feasibility_dut src/feasibility_dut.c src/capture_program.c)
        add_dependencies(feasibility_dut bridge_validate)
        target_include_directories(feasibility_dut PRIVATE src "${DEPS}/apio/include")
        target_link_libraries(feasibility_dut PRIVATE pico_stdlib hardware_pio)
        target_compile_options(feasibility_dut PRIVATE -Wall -Wextra -Werror)
        if(DUT_CAPTURE_W1)
            target_compile_definitions(bridge_capture PRIVATE CAPTURE_W1=1)
            target_compile_definitions(feasibility_dut PRIVATE CAPTURE_W1=1)
        endif()
        pico_enable_stdio_uart(feasibility_dut 0)
        pico_enable_stdio_usb(feasibility_dut 1)
        pico_set_binary_type(feasibility_dut no_flash)
        pico_add_extra_outputs(feasibility_dut)
    endif()
endif()
